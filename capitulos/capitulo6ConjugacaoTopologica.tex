	\chapter{Conjugação Topológica}
	
	O objetivo deste capítulo é relacionar a aplicação \textit{shift} discutida anteriormente com o mapa logístico $F_\mu$, quando $\mu > 2 + \sqrt{5}$. Lembre-se de que com exceção dos pontos no conjunto de Cantor $\Lambda$, todos os outros pontos em $\mathbb{R}$ tendem a $-\infty$ sob a iteração de $F_\mu$. Para completar a descrição da dinâmica de $F_\mu$, devemos então compreender a restrição de $F_\mu$ em $\Lambda$.
	
	\section{O Mapa Itinerário}
	
	Temos que se $x\in \Lambda \subset I_0 \cup I_1$, então todos pontos na órbita de $x$ também estão em $\Lambda$, e com isso estão em um desses dois intervalos. Podemos, desse modo, obter uma ideia aproximada do comportamento da órbita observando em qual desses intervalos caem as várias iteradas de $x$. Assim, estabelecemos a seguinte definição.
	
	\begin{definicao}
		O itinerário de $x$ é a sequência $S(x) = s_0s_1s_2...$ onde $s_j = 0$ se $F_\mu^j(x) \in I_0$ e $s_j = 1$ se $F_\mu^j(x) \in I_1$, para cada $x \in \Lambda$.
	\end{definicao}
	
	Assim, o itinerário de $x$ é uma sequência infinita de 0s e 1s. Ou seja, $S(x)$ é um ponto em $\Sigma_2$, isto é, o itinerário é uma aplicação de $\Lambda$ em $\Sigma_2$.
	
	\begin{teorema}
		Se $\mu > 2+\sqrt{5}$, então o itinerário $S: \Lambda \rightarrow \Sigma_2$ é um homeomorfismo.
	\end{teorema}
	
	\begin{prova}
		\begin{itemize}
			\item $S$ é injetora:
			
			Sejam $x,y \in \Lambda$, com $x<y$. Supondo $S(x) = S(y) = s_0s_1... \in \Sigma_2$, temos que $F_\mu^n(x), F_\mu^n(y) \in I_{s_n}, \forall n\geq0$, isto é, $F^n(x) e F^n(y)$ estarão simultaneamente na mesma metade do intervalo $[0,1]$. Isso implica que $F$ é monótona no intervalo com extremidades $F^n(x)$ e $F^n(y)$ e, portanto, teríamos esse intervalo contido em $I_0 \cup I_1$ para todo $n$, isto é, $[x,y]\subset \Lambda$.
			Mas isso contradiz o fato de $\Lambda$ não possuir intervalos, portanto $S$ é injetora.
			
			\item $S$ é sobrejetora:
			
			Primeiramente, observe que se $J \subset I$ é um intervalo fechado, então $F^{-1}(J)$ consiste de 2 subintervalos, um em $I_0$ e outro em $I_1$. Ver Figura...
			
			Dado $s=(s_0s_1s_2...)\in\Sigma_2$, vamos encontrar $x\in\Lambda$ tal que $S(x)=s$. Para isso, defina
			$$I_{s_0s_1...s_n}=\{x\in I : x \in I_{s_0}, F(x) \in I_{s_1},...,F^n(x) \in I_{s_n}\} = I_{s_0} \cap F^{-1}(I_{s_1}) \cap ... \cap F^{-n}(I_{s_n}).$$
			
			Assim, temos que $I_{s_0s_1...s_n}$ é uma sequência de intervalos fechados não-vazios. Note que
			$$I_{s_0s_1...s_n} = I_{s_0} \cap F^{-1}(I_{s_1...s_n})$$
			e, por indução, $I_{s_1...s_n}$ é um subintervalo não-vazio. Pelo observado acima, $F^{-1}(I_{s_1...s_n})$ consiste de dois intervalos fechados, um em $I_O$ e outro em $I_1$ e, portanto, $I_{s_0s_1...s_n} = I_{s_0} \cap F^{-1}(I_{s_1...s_n})$ é um intervalo fechado. Além disso, esses intervalos são encaixantes, pois 
			$$I_{s_0...s_n} = I_{s_0...s_{n-1}} \cap F^{-n}(I_{s_n}) \subset I_{s_0...s_{n-1}}.$$
			Consequentemente, pelo Teorema dos Intervalos Encaixantes, temos que 
			$$\bigcap_{n \geq 0} I_{s_0...s_n} \neq \emptyset.$$
			Por fim, note que se $x \in \bigcap_{n \geq 0} I_{s_0...s_n}$, então 
			$$x \in I_{s_0}, F(x) \in I_{s_1}, F^2(x) \in I_{s_2}, ..., F^n(x) \in I_{s_n},...,$$
			isto é, $S(x) = (s_0s_1...)$. Isso prova que $S$ é sobrejetora.
			
			\item $S$ é contínua:
			
			Tomemos $x \in \Lambda$ e supomos que $S(x) = s_0s_1s_2 \dots$. Seja $\epsilon > 0$ e $n$ tal que $\frac{1}{2^n} < \epsilon$. Considere os subintervalos fechados $I_{t_0t_1\dots t_n}$ para todas as combinações possíveis $t_0t_1 \dots t_n$. 
			
			Estes subintervalos são todos disjuntos, e $\Lambda$ está contido na união deles. Como existem $2^{n+1}$ subintervalos, logo $I_{s_0s_1\dots s_n}$ é um deles. Portanto, podemos escolher $\delta$ tal que $|x - y| < \delta$ e $y \in \Lambda$ implique que $y \in I_{s_0s_1\dots s_n}$. 
			
			Assim, $S(y)$ coincide com $S(x)$ nos primeiros $n + 1$ termos. Portanto, pelo Teorema da Proximidade:
			$$ d[S(x), S(y)] < \frac{1}{2^n} < \epsilon $$
			
			Isso prova a continuidade de $S$.
			
			\item A inversa $S^{-1}$ contínua:
			
			Seja $s = s_0 s_1 \dots \in \Sigma_2$ e $\epsilon > 0$. Sabemos pela construção do conjunto de Cantor $\Lambda$ que o comprimento dos subintervalos da forma $I_{s_0 \dots s_n}$ tende a zero quando $n \to \infty$. Assim, podemos tomar $n$ tal que o comprimento do intervalo  $I_{s_0...s_n}$ seja menor que $\epsilon$, isto é,
			$$|I_{s_0 \dots s_n}| < \epsilon.$$
			
			Escolhemos $\delta = \frac{1}{2^n}$. Seja $t \in \Sigma_2$ uma sequência qualquer tal que $d[s, t] < \delta$. Pela definição da métrica em $\Sigma_2$, isso garante que as sequências $s$ e $t$ coincidem em seus primeiros $n+1$ termos. 
			
			Pela definição da aplicação de itinerário, se $s$ e $t$ começam com o bloco $s_0 \dots s_n$, então tanto $S^{-1}(s)$ quanto $S^{-1}(t)$ pertencem ao mesmo $I_{s_0 \dots s_n}$. 
			Como $s$ e $t$ estão no mesmo intervalo, a distância entre eles não pode ser maior que o comprimento do mesmo.
			$$ |S^{-1}(s) - S^{-1}(t)| \le |I_{s_0 \dots s_n}| < \epsilon $$
			então,
			$$|s-t|<\frac{1}{2^n} \implies |S^{-1}(s) - S^{-1}(t)| \le |I_{s_0 \dots s_n}| < \epsilon.$$
			Com isso, $S^{-1}$ é contínua.
		\end{itemize}
		
		Portanto, $S$ é de fato um homeomorfismo.
	\end{prova}
	
	O teorema acima mostra que, como conjuntos, $\Lambda$ e $\Sigma_2$ são os mesmos. Mais importante, a aplicação $S$ dá uma equivalência entre as dinâmicas de $F$ e $\simga$.
	
	\begin{definicao}
		Sejam $f: A \rightarrow A$ e $g: B \rightarrow B$ duas funções. Dizemos que $f$ e $g$ são topologicamente conjugadas se existe um homeomorfismo $h: A \rightarrow B$ tal que $h \circ f = g \circ h$. A equação $h \circ f = g \circ h$ é dita equação de conjugação e o homeomorfismo $h$ é dito conjugação topológica.
	\end{definicao}
	
	\begin{exemplo}
		Se $c<\frac{1}{4}$ e $\mu>1$, então o mapa quadrático $Q_c$ e o mapa logístico $F_\mu$ são topologicamente conjugados por um mapa linear do tipo $ax+b$. Ou seja,
		$$
		\begin{array}{ccc}
			x & \xrightarrow{\quad F_\mu \quad} & \mu\/x(1 - x) \\
			h \Bigg\downarrow & & \Bigg\downarrow h \\
			ax + b & \xrightarrow{\quad Q_c \quad} & \boldsymbol{(ax+b)^2+c = a(\mu\/x(1-x))+b}
		\end{array}
		$$
		
		Desenvolvendo a igualdade de polinômios, teremos $a=-\mu, b=\frac{\mu}{2}$ e $b^2+c = b$. Para garantir $\mu > 1$, basta tomar $c<\frac{1}{4}$. Portanto, a igualdade é válida, fazendo $Q_c$ e $F_\mu$ serem topologicamente conjugados por um mapa linear.
	\end{exemplo}
	
	Mapas que são topologicamente conjugados são completamente equivalentes em termos de dinâmica. Pontos periódicos e fixos, órbitas e densidade são preservados na transição. Desse modo, estudar a dinâmica em $g$ é a mesma coisa que estudar a dinâmica em $f$.
	
	\vspace{-\baselineskip}
	No caso do mapa shift $\sigma$ em $\Sigma_2$ e $F_\mu$ em $\Lambda$, o próximo Teorema nos mostrará que esses dois mapas são topologicamente conjugados, fazendo com que do ponto de vista dinâmico, se tornem iguais. Portanto, poderemos continuar estudando a dinâmica de $F_\mu$ por meio de $\sigma$, o que se mostrará bem mais simples de ser feito.
	
	\begin{teorema}
		$S \circ F_\mu = \sigma \circ S$.
	\end{teorema}
	
	\begin{prova}
		Um ponto $x$ em $\Lambda$ pode ser definido pela sequência aninhada de intervalos
		$$ \bigcap_{n \ge 0} I_{s_0 s_1 \dots s_n} $$
		determinada pelo itinerário $S(x)$. Por definição, sabemos que:
		$$ I_{s_0 \dots s_n} = I_{s_0} \cap F_\mu^{-1}(I_{s_1}) \cap F_\mu^{-2}(I_{s_2}) \cap \dots \cap F_\mu^{-n}(I_{s_n}) $$
		
		Aplicando a função $F_\mu$ a ambos os lados da igualdade, temos
		$$ F_\mu(I_{s_0 \dots s_n}) = F_\mu(I_{s_0}) \cap F_\mu(F_\mu^{-1}(I_{s_1})) \cap F_\mu(F_\mu^{-2}(I_{s_2})) \cap \dots \cap F_\mu(F_\mu^{-n}(I_{s_n})) $$
		$$ F_\mu(I_{s_0 \dots s_n}) = I \cap I_{s_1} \cap F_\mu^{-1}(I_{s_2}) \cap \dots \cap F_\mu^{-n+1}(I_{s_n}) $$
		
		Note que essa expressão é exatamente a definição de $I_{s_1 \dots s_n}$. Portanto, 
		$$ S(F_\mu(x)) = S \left( F_\mu \left( \bigcap_{n=0}^{\infty} I_{s_0 s_1 \dots s_n} \right) \right) = S \left( \bigcap_{n=1}^{\infty} I_{s_1 \dots s_n} \right) = s_1 s_2 \dots = \sigma(S(x)). $$
	\end{prova}
	
	Em particular, como $F_\mu$ em $\Lambda$ é topologicamente conjugada ao mapa shift em $\Sigma$, concluímos que $F_\mu$ possui as mesmas propriedades que descobrimos anteriormente para $\sigma$. Isso pode ser resumido da seguinte forma:
	
	\begin{teorema}
		Seja $F_\mu(x) = \mu x(1 - x)$ com $\mu > 2 + \sqrt{5}$. Então:
		\begin{enumerate}
			\item A cardinalidade de $Per_n(F_\mu)$ é $2^n$.
			\item $Per(F_\mu)$ é denso em $\Lambda$.
			\item $F_\mu$ possui uma órbita densa em $\Lambda$.
		\end{enumerate}
	\end{teorema}
	
	Esse teorema mostra o poder da dinâmica simbólica e da conjugação topológica. Realizar o estudo da dinâmica pelo mapa logístico se torna rapidamente uma tarefa algebricamente inviável, mas a conjugação topológica garante que é possível estudar toda essa dinâmica por um mapa equivalente e mais simples.
	
	Outra ferramenta para relacionar a dinâmica de dois mapas diferentes é a semi-conjugação.
	
	\begin{definicao}
		Sejam $f : X \to X$ e $g : Y \to Y$, e suponha que $h : X \to Y$ seja uma aplicação contínua e sobrejetora de $X$ em $Y$ que satisfaz a equação de conjugação
		$$ h(f(x)) = g(h(x)) $$
		para todo $x \in X$. Então $h$ é uma semi-conjugação, e as aplicações $f$ e $g$ são semi-conjugadas entre si.
	\end{definicao}
	
	\noindent\textbf{Observação:} Diferentemente da conjugação, a semi-conjugação não exige que $h$ seja injetora (logo, não é um homeomorfismo). Além disso, as semi-conjugações preservam órbitas densas e levam pontos periódicos de $f$ em pontos periódicos de $g$, embora o período possa ser menor (por exemplo, um ciclo de período 2 pode ser mapeado em um único ponto fixo).
	
	\begin{exemplo}
		Seja $D : S^1 \to S^1$ a aplicação de duplicação no círculo unitário, dada por $D(\theta) = 2\theta$. Assim, $D$ simplesmente duplica o ângulo de qualquer ponto que esteja em $S^1$. E seja $Q_{-2}(x) = x^2 - 2$ onde $x \in [-2, 2]$. Então seja $h : S^1 \to [-2, 2]$ dada por $h(\theta) = 2\cos(\theta)$. Note que $h$ projeta o círculo de forma "dois-para-um" sobre $[-2, 2]$ (exceto nos dois pontos onde o círculo intercepta a reta real).  Ou seja, dois pontos gerados pelo ângulo $\theta$ acabam sendo levados em um único ponto na reta real.
		
		\begin{center}
			\begin{tikzpicture}[scale=1.3]
				% Eixos X e Y
				\draw[thick, ->] (-3, 0) -- (3, 0) node[right] {$x$};
				\draw[thick, ->] (0, -3) -- (0, 3) node[above] {$y$};
				
				% Círculo (S1) de raio 2 centrado na origem
				\draw[thick] (0,0) circle (2);
				
				
				
				% Definindo a origem e o ângulo (ex: 50 graus para não sobrepor o '1')
				\coordinate (O) at (0,0);
				\def\ang{50} 
				
				% Coordenadas dos pontos no círculo de raio 2
				\coordinate (P1) at ({\ang}:2);
				\coordinate (P2) at ({-\ang}:2);
				\coordinate (Px) at ({2*cos(\ang)}, 0);
				
				% Ligações do centro (0,0) aos pontos
				\draw[thick] (O) -- (P1);
				\draw[thick] (O) -- (P2);
				
				% Desenhando os arcos dos ângulos \theta e -\theta
				\draw[->, thick, green] (0.6, 0) arc (0:\ang:0.6) node[midway, right=1pt, yshift=1pt] {$\theta$};
				\draw[->, thick, green] (0.6, 0) arc (0:-\ang:0.6) node[midway, right=-1pt, yshift=-1pt] {$-\theta$};
				
				% Projeções verticais (função h esmagando no eixo x)
				\draw[->, dashed, red, thick] (P1) -- (Px);
				\draw[->, dashed, red, thick] (P2) -- (Px);
				
				% Marcando os pontos e colocando as legendas de coordenadas
				\filldraw (P1) circle (1.5pt) node[above right] {$(2\cos\theta, 2\sin\theta)$};
				\filldraw (P2) circle (1.5pt) node[below right] {$(2\cos\theta, -2\sin\theta)$};
				
				% Ponto projetado no eixo x
				\filldraw[green] (Px) circle (1.5pt);
				\node[above right, text=black, yshift=2pt] at (Px) {$h(\theta) = 2\cos\theta$};
				
				% Rótulo da função de projeção
				\node[red, right] at ({2*cos(\ang)}, {sin(\ang)}) {$h$};
				
			\end{tikzpicture}
			\captionof{figure}{Ilustração da semi-conjugação.}
		\end{center}
		
		$h$ é uma semi-conjugação entre essas duas aplicações, pois
		$$ h(D(\theta)) = 2\cos(2\theta) = 2(\cos^2(\theta) - \sin^2(\theta)) = 2(2\cos^2(\theta) - 1) = 4\cos^2(\theta) - 2. $$
		Por outro lado, temos:
		$$ Q_{-2}(h(\theta)) = (2\cos(\theta))^2 - 2 = 4\cos^2(\theta) - 2 = h(D(\theta)). $$
		Portanto, essas aplicações de fato são semi-conjugadas.
	\end{exemplo}
